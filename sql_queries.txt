Launch count by site
SELECT LaunchSite, COUNT(*) AS launches FROM launches GROUP BY LaunchSite ORDER BY launches DESC

Success rate by site
SELECT LaunchSite, COUNT(*) AS launches, ROUND(100.0*AVG(Class),2) AS success_rate_pct FROM launches GROUP BY LaunchSite ORDER BY success_rate_pct DESC

Success rate by orbit
SELECT Orbit, COUNT(*) AS launches, ROUND(100.0*AVG(Class),2) AS success_rate_pct FROM launches GROUP BY Orbit ORDER BY success_rate_pct DESC

Payload summary
SELECT Orbit, ROUND(AVG(PayloadMass),2) AS avg_payload_kg, MIN(PayloadMass) AS min_payload_kg, MAX(PayloadMass) AS max_payload_kg FROM launches GROUP BY Orbit ORDER BY avg_payload_kg DESC

Top flight records
SELECT FlightNumber, LaunchSite, Orbit, PayloadMass, Class FROM launches ORDER BY FlightNumber DESC LIMIT 10
